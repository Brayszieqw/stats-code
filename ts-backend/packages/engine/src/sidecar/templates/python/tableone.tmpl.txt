import pandas as pd
from scipy import stats

data = pd.read_csv("data.csv")
group = {{params.group_quoted}}
continuous = [{{params.continuous_quoted}}]
groups = data[group].dropna().unique()
for variable in continuous:
    group_a = data[data[group] == groups[0]][variable]
    group_b = data[data[group] == groups[1]][variable]
    result = stats.ttest_ind(group_a, group_b)
